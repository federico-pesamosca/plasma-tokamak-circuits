# All commands from the repository root. Results are saved in the repository root
# (plasmaless_*_results.mat, ignored by git); the machine description is data/md_dd4.
# Plotting hangs in MATLAB R2025b on the SDCC login nodes: do_plot = 0 by default,
# plot afterwards with R2024b (last section).

# Within Matlab, IMAS, without simulink
module purge
module load IMAS-MATLAB/5.6.0-intel-2025b-DD-4.1.1
export OMP_NUM_THREADS=1
matlab
maxNumCompThreads(2);   % optional, keeps it within the login-node guideline
run_plasmaless_imas      % --> plasmaless_imas_results.mat

# Simulink equivalent:
module purge
module load IMAS-MATLAB/5.6.0-intel-2025b-DD-3.42.2
export OMP_NUM_THREADS=1
matlab
maxNumCompThreads(2);
run_plasmaless_model     % --> plasmaless_simulink_results.mat (do_plot = 1 before: original plots)

# Muscle3 equivalent (dt 0.01)
cd muscle3
./run_plasmaless_muscle3.sh # --> plasmaless_muscle3_results.mat in the repository root
                            #     (MUSCLE3 logs in /scratch/users/$USER/plasmaless/m3_standalone_<date>)
cd ..

# Plotting: set do_plot = 1 before the run (plot_plasmaless_model is called at the end of
# run_plasmaless_imas / run_plasmaless_model; it reads the machine description in data/md_dd4,
# so the IMAS-MATLAB DD-4.1.1 module is needed)
do_plot = 1; run_plasmaless_imas
do_plot = 1; run_plasmaless_model
